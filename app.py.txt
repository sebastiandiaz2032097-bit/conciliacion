import streamlit as st
import pandas as pd

st.set_page_config(page_title="Conciliación Contable", layout="wide")

st.title("📊 Portal de Conciliación: Balance General vs. Amortización")
st.write("Sube tus archivos planos (CSV) o de Excel para cruzar saldos automáticamente.")

# Sección de carga de archivos
col1, col2 = st.columns(2)

with col1:
    st.subheader("1. Balance General por Cuentas")
    file_bg = st.file_uploader("Cargar Balance (CSV/Excel)", type=["csv", "xlsx"])

with col2:
    st.subheader("2. Auxiliar de Amortización")
    file_amort = st.file_uploader("Cargar Amortizaciones (CSV/Excel)", type=["csv", "xlsx"])

# Procesamiento cuando ambos archivos están subidos
if file_bg and file_amort:
    # Leer archivos
    df_bg = pd.read_excel(file_bg) if file_bg.name.endswith('.xlsx') else pd.read_csv(file_bg)
    df_amort = pd.read_excel(file_amort) if file_amort.name.endswith('.xlsx') else pd.read_csv(file_amort)

    st.success("¡Archivos cargados correctamente!")

    # Mostrar vistas previas
    st.write("---")
    c1, c2 = st.columns(2)
    with c1:
        st.write("**Vista previa: Balance General**")
        st.dataframe(df_bg.head())
    with c2:
        st.write("**Vista previa: Amortización**")
        st.dataframe(df_amort.head())
